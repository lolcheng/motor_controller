cmake_minimum_required(VERSION 3.16)
project(encos_driver VERSION 0.0.1 LANGUAGES CXX)
include(CTest)
find_package(ament_cmake REQUIRED)
find_package(Threads REQUIRED)
find_package(encos_core REQUIRED)
find_package(encos_msgs REQUIRED)
find_package(encos_experiments REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
add_library(driver_runtime STATIC src/ros_interfaces.cpp src/runtime_config.cpp src/control_worker.cpp src/motor_node.cpp)
target_compile_features(driver_runtime PUBLIC cxx_std_17)
target_compile_options(driver_runtime PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(driver_runtime PUBLIC include)
target_link_libraries(driver_runtime PUBLIC encos_core::encos_core encos_experiments::trajectory Threads::Threads)
ament_target_dependencies(driver_runtime PUBLIC encos_msgs rclcpp rclcpp_action sensor_msgs rcl_interfaces)
add_executable(encos_driver_node src/main.cpp)
target_link_libraries(encos_driver_node PRIVATE driver_runtime)
install(TARGETS encos_driver_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME} FILES_MATCHING PATTERN "*.yaml" PATTERN "*.launch.py")
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(test_ros_interfaces test/test_ros_interfaces.cpp)
  target_link_libraries(test_ros_interfaces PRIVATE driver_runtime)
  add_test(NAME p2_interfaces COMMAND test_ros_interfaces)
  add_test(NAME p2_vcan_ros COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test/test_driver_launch.py"
           "$<TARGET_FILE:encos_driver_node>")
  set_tests_properties(p2_vcan_ros PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 180)
endif()
ament_package()

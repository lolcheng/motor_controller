cmake_minimum_required(VERSION 3.16)
project(encos_driver VERSION 0.0.1 LANGUAGES CXX)
include(CTest)
find_package(ament_cmake REQUIRED)
find_package(encos_core REQUIRED)
find_package(encos_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)

add_library(driver_runtime STATIC src/ros_interfaces.cpp)
target_compile_features(driver_runtime PUBLIC cxx_std_17)
target_compile_options(driver_runtime PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(driver_runtime PUBLIC include)
target_link_libraries(driver_runtime PUBLIC encos_core::encos_core)
ament_target_dependencies(driver_runtime PUBLIC encos_msgs rclcpp rcl_interfaces)

add_executable(encos_usb_can_node src/usb_can_node.cpp)
target_link_libraries(encos_usb_can_node PRIVATE driver_runtime)
target_compile_options(encos_usb_can_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS encos_usb_can_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME} FILES_MATCHING PATTERN "*.launch.py")

if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(test_ros_interfaces test/test_ros_interfaces.cpp)
  target_link_libraries(test_ros_interfaces PRIVATE driver_runtime)
  add_test(NAME ros_interfaces COMMAND test_ros_interfaces)
  add_test(NAME usb_can_ros_offline COMMAND "${Python3_EXECUTABLE}"
           "${CMAKE_CURRENT_SOURCE_DIR}/test/test_usb_can_ros.py" "$<TARGET_FILE:encos_usb_can_node>")
  set_tests_properties(usb_can_ros_offline PROPERTIES TIMEOUT 30)
endif()
ament_package()

cmake_minimum_required(VERSION 3.16)
project(encos_experiments VERSION 0.0.1 LANGUAGES CXX)
include(CTest)
find_package(ament_cmake REQUIRED)
add_library(trajectory src/trajectory.cpp)
target_compile_features(trajectory PUBLIC cxx_std_17)
target_compile_options(trajectory PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(trajectory PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
install(TARGETS trajectory EXPORT encos_experiments_targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include FILES_MATCHING PATTERN "trajectory.hpp")
install(PROGRAMS scripts/export_samples.py scripts/analyze_timing.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(test_trajectory test/test_trajectory.cpp)
  target_link_libraries(test_trajectory PRIVATE trajectory)
  add_test(NAME trajectory COMMAND test_trajectory)
endif()
ament_export_targets(encos_experiments_targets HAS_LIBRARY_TARGET)
ament_package()
